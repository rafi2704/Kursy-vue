<template>
  <div>
    <h1>Lista Postów</h1>
    <ul>
      <!-- Wyświetlanie postów -->
      <li v-for="post in posts" :key="post.id">
        <h2>{{ post.title }}</h2>
        <p>{{ post.body }}</p>
      </li>
    </ul>

    <!-- Komunikat o błędzie -->
    <div v-if="error" style="color: red">
      <p>Wystąpił błąd podczas pobierania danych!</p>
    </div>

    <!-- Komunikat o ładowaniu -->
    <div v-if="loading">
      <p>Ładowanie...</p>
    </div>
  </div>
</template>

<script>
import axios from "axios";

export default {
  data() {
    return {
      posts: [], // Tablica na posty
      loading: false, // Flaga ładowania
      error: false, // Flaga błędu
    };
  },
  mounted() {
    this.fetchPosts(); // Wywołanie funkcji przy załadowaniu komponentu
  },
  methods: {
    // Funkcja pobierająca dane z API
    async fetchPosts() {
      this.loading = true;
      this.error = false;

      try {
        const response = await axios.get(
          "https://jsonplaceholder.typicode.com/posts",
        );
        this.posts = response.data; // Zapisujemy odpowiedź do zmiennej posts
      } catch (err) {
        this.error = true; // W przypadku błędu ustawiamy flagę błędu
      } finally {
        this.loading = false; // Ustawiamy flagę ładowania na false po zakończeniu zapytania
      }
    },
  },
};
</script>
